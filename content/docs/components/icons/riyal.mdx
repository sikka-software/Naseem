## Usage

```tsx
import { RiyalIcon } from "@/components/naseem-ui/icons/riyal"

export default function Example() {
  return (
    <RiyalIcon className="h-12 w-12" />
  )
}
```

You can also import directly:

```tsx
import { RiyalIcon } from "@/components/naseem-ui/icons/riyal"
```

## Examples

### Icon Sizes

```tsx
<div className="flex items-center gap-4">
  <RiyalIcon className="h-4 w-4" />  {/* Small */}
  <RiyalIcon className="h-6 w-6" />  {/* Default */}

```
